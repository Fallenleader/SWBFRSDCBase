# Native engine (docs/NATIVE_ENGINE.md).  Independent of the decompiled game: see tools/check_independence.sh.
#   desktop:  cmake -S engine -B build/engine -DCMAKE_BUILD_TYPE=Release && cmake --build build/engine
#   Vita / 3DS / PS5 toolchains join with their platform back ends (milestone 1.2).
cmake_minimum_required(VERSION 3.16)
project(aurek C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# every performance flag (owner's rule); per-device CPU flags come from AK_TARGET
set(AK_TARGET "native" CACHE STRING "desktop CPU: native | x86_64_v3 (Windows/PC) | znver2 (Steam Deck)")
set(AK_OPT -O3 -fomit-frame-pointer -ffast-math -fno-finite-math-only -fno-math-errno -ffp-contract=fast -fno-signed-zeros -fno-trapping-math
           -funroll-loops -ftree-vectorize -fvect-cost-model=dynamic -fpredictive-commoning -fgcse-after-reload -ftree-loop-distribution
           -floop-nest-optimize -fgraphite-identity -fipa-pta -fipa-cp-clone -fsched-pressure -fno-semantic-interposition
           -fmerge-all-constants -fno-common -fno-plt -fno-stack-protector -ffunction-sections -fdata-sections)
if(AK_TARGET STREQUAL "znver2")
  list(APPEND AK_OPT -march=znver2 -mtune=znver2)
elseif(AK_TARGET STREQUAL "x86_64_v3")
  list(APPEND AK_OPT -march=x86-64-v3 -mtune=generic)
else()
  list(APPEND AK_OPT -march=native -mtune=native)
endif()
add_compile_options(-Wall -Wextra -Wno-unused-parameter "$<$<CONFIG:Release>:${AK_OPT}>")
include(CheckIPOSupported)
check_ipo_supported(RESULT AK_LTO OUTPUT _ipo_msg)
if(AK_LTO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  add_link_options("$<$<CONFIG:Release>:-flto-partition=one;-fuse-linker-plugin;-fdevirtualize-at-ltrans;-Wl,-O1;-Wl,--sort-common>")
endif()

add_custom_target(independence ALL COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_independence.sh VERBATIM)

add_library(akcore STATIC core/ak_core.c core/ak_vfs.c core/ak_settings.c assets/asura.c assets/model.c assets/entity.c assets/blueprint.c assets/gui.c game/ak_soldier.c game/ak_character.c game/ak_weapon.c game/ak_conquest.c game/ak_ctf.c game/ak_gc.c game/ak_gc_ai.c game/ak_script.c game/ak_vehicle.c game/ak_space.c game/ak_turret.c game/ak_projectile.c game/ak_pfx.c game/ak_path.c game/ak_loco.c world/ak_collision.c world/ak_nav.c render/ak_scene.c audio/ak_audio.c audio/ak_music.c audio/ak_speech.c audio/ak_adp.c video/ak_thp.c)
# The GC AI ranks plans by float scores that must round as the PSP's single-precision code does (no fused multiply-add)
set_source_files_properties(game/ak_gc_ai.c PROPERTIES COMPILE_OPTIONS "-fno-fast-math;-ffp-contract=off")
target_include_directories(akcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(akcore PUBLIC m)

add_library(akplat_desktop STATIC platform/desktop/plat_desktop.c)
target_link_libraries(akplat_desktop PUBLIC akcore)
find_package(Threads REQUIRED)
target_link_libraries(akplat_desktop PUBLIC Threads::Threads)

add_executable(akinspect tools/akinspect.c)
target_link_libraries(akinspect PRIVATE akcore akplat_desktop)
target_link_options(akinspect PRIVATE -Wl,--gc-sections)
add_executable(akbp tools/akbp.c)
target_link_libraries(akbp PRIVATE akcore akplat_desktop)

# the game (milestone 1.2): the desktop renderer is OpenGL 3.3 (development, headless checks) or Vulkan 1.3 (Windows, Steam Deck)
set(AK_RENDERER "gl" CACHE STRING "desktop renderer: gl | vulkan")
find_package(SDL2 REQUIRED)
execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" describe --always --dirty=+ OUTPUT_VARIABLE AK_GIT_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
set(AK_GAME_SRC app/viewer.c app/play.c app/play_sound.c app/hud.c app/frontend.c app/gc_screen.c video/ak_movie.c platform/desktop/plat_sdl.c)
if(AK_RENDERER STREQUAL "vulkan")
  find_program(GLSLANG glslangValidator REQUIRED)
  set(AK_VK_DIR ${CMAKE_CURRENT_SOURCE_DIR}/platform/desktop/vk)
  file(GLOB AK_VK_SHADERS ${AK_VK_DIR}/*.vert ${AK_VK_DIR}/*.frag)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gfx_vk_spv.h
    COMMAND ${CMAKE_COMMAND} -DSRC=${AK_VK_DIR} -DOUT=${CMAKE_CURRENT_BINARY_DIR}/gfx_vk_spv.h -DGLSLANG=${GLSLANG} -P ${AK_VK_DIR}/spv.cmake
    DEPENDS ${AK_VK_SHADERS} ${AK_VK_DIR}/spv.cmake VERBATIM)
  add_executable(aurek_viewer ${AK_GAME_SRC} platform/desktop/gfx_vk.c ${CMAKE_CURRENT_BINARY_DIR}/gfx_vk_spv.h)
  set_source_files_properties(platform/desktop/gfx_vk.c PROPERTIES COMPILE_OPTIONS -Wno-missing-field-initializers)
  target_include_directories(aurek_viewer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  if(AK_VULKAN_INCLUDE)
    target_include_directories(aurek_viewer PRIVATE ${AK_VULKAN_INCLUDE})
  endif()
  set(AK_GFX_LIBS "")                         # (no link to Vulkan: SDL loads vulkan-1.dll / libvulkan.so.1 at run time)
else()
  find_package(OpenGL REQUIRED)
  add_executable(aurek_viewer ${AK_GAME_SRC} platform/desktop/gfx_gl.c)
  set(AK_GFX_LIBS OpenGL::GL)
endif()
target_compile_definitions(aurek_viewer PRIVATE AK_BUILD_ID="${AK_GIT_ID}")
target_include_directories(aurek_viewer PRIVATE ${SDL2_INCLUDE_DIRS})
if(TARGET SDL2::SDL2-static AND AK_STATIC_SDL)
  target_link_libraries(aurek_viewer PRIVATE akcore akplat_desktop SDL2::SDL2-static ${AK_GFX_LIBS} m)
else()
  target_link_libraries(aurek_viewer PRIVATE akcore akplat_desktop ${SDL2_LIBRARIES} ${AK_GFX_LIBS} m)
endif()
target_link_options(aurek_viewer PRIVATE -Wl,--gc-sections)
# the movies (engine/video/ak_movie.c): the THP conversion always; with libavcodec also the Vita conversion's MP4 / the game's PMF
option(AK_LIBAV "decode the MP4 / PMF movies with libavcodec where found" ON)
if(AK_LIBAV)
  find_package(PkgConfig QUIET)
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(AKAV QUIET IMPORTED_TARGET libavformat libavcodec libavutil)
  endif()
  if(AKAV_FOUND)
    target_compile_definitions(aurek_viewer PRIVATE AK_HAVE_LIBAV=1)
    target_link_libraries(aurek_viewer PRIVATE PkgConfig::AKAV)
    message(STATUS "movies: libavcodec ${AKAV_libavcodec_VERSION} (MP4 / PMF) and THP")
  else()
    message(STATUS "movies: THP only (no libavcodec)")
  endif()
endif()
if(WIN32)
  target_link_options(aurek_viewer PRIVATE -static -static-libgcc -mwindows)
  set_target_properties(aurek_viewer PROPERTIES OUTPUT_NAME Aurek)
endif()
