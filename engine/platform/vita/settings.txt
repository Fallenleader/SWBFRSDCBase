# Aurek level viewer (PS Vita) settings: copy to ux0:data/Aurek/settings.txt
# One KEY=VALUE per line; '#' starts a comment.  Every value below is the default, so deleting a line changes nothing.
#
# Game files: game.iso or GAME/PSP_GAME/... in ux0:data/Aurek, or (if not there) in ux0:data/SWBFRS, where the earlier port keeps
# them.  Nothing needs copying if the earlier port already runs.  ur0:data/libshacccg.suprx is needed once, to compile the shaders
# at first start (cached in ux0:data/Aurek/shaders afterwards).
#
# Controls: left stick move, right stick look, L / R down / up, hold Cross to move faster,
#           START next level, TRIANGLE previous level, SELECT figures on/off.  Log: ux0:data/Aurek/log.txt (one line per second).
#           CIRCLE: model gallery (every model of the level in its bind pose): d-pad left / right previous / next model,
#           right stick turns, left stick zooms, CIRCLE again back to the level.
#           SQUARE: play (a soldier in a conquest match with bots) / free camera.  Playing: left stick move, right stick aim
#           (twin stick), CROSS jump, R fire, L hold lock-on, TRIANGLE get in / out of a vehicle; START changes level (a new match
#           starts; TRIANGLE goes back a level only in the free camera).  Driving: left stick throttle / steer, L held strafe, CROSS boost.
#           Flying: left stick pitch (forward = nose down; AK_INVERT_Y flips it) and bank, CROSS boost, L held slow down, R fire; any
#           input on the ground takes off, touching down slowly and level lands.  Bots take free vehicles of their side too.

# ---- start
# level shown first, as "<ERA> <MAP>" (eras: CLASSIC, PREQUEL, CAMPAIGN)
AK_LEVEL=CLASSIC HOTH
# figures on screen at start (fps, frame time, draws, triangles, GPU wait)
AK_SHOW_FIGURES=1
# 1 = start in the model gallery
AK_GALLERY=0

# ---- picture
# 1 = wait for the display's refresh (no tearing, capped at 60 fps); 0 = uncapped, to measure how fast frames really are
AK_VSYNC=1
# clipping distances in metres (space levels have ships and planets several km away)
AK_NEAR=0.75
AK_FAR=8000
# vertex colour scale: the game's baked vertex colours use 128 = full brightness, so 2 is the game's look
AK_COLOUR_SCALE=2.0
# distance fog in metres (end <= start: none)
AK_FOG_START=0
AK_FOG_END=0
# back-face culling of solid level surfaces: 0 = off (as the game draws: some surfaces are single-sided panels meant to be seen
# from both sides), 1 = on (holes appear where such panels are seen from behind; the Vita's GPU removes hidden surfaces anyway,
# so culling saves little), 2 = cull the other side.  See-through surfaces and models are always drawn from both sides.
AK_CULL=0

# ---- memory
# video memory for one level's geometry and textures, in MB
AK_VRAM_POOL_MB=64
# main memory for one level's data, in MB
AK_LEVEL_MB=48

# --- play mode (square in the level view toggles it) ---
# start in play mode: 1 = a soldier on foot, 0 = free camera
AK_PLAY=1
# soldiers' size relative to the world: 1.0 = the game's own (the models are authored at 1.75 m); body, hit shapes and camera scale with it
AK_SOLDIER_SCALE=1.0
# the soldier class (a Humanoid blueprint: Stormtrooper, MediumRebel, Clonetrooper, Battle_Droid, Darktrooper, ...)
AK_SOLDIER=Stormtrooper
# head 0..2, personalisation colours 0..15 (-1 = the faction's default)
AK_HEAD=0
AK_COLOUR1=-1
AK_COLOUR2=-1
# weapon (the faction's own version is used): BlasterRifle, BlasterPistol, Sniper Rifle, ...
AK_WEAPON=BlasterRifle
# bots in all, shared between the sides (the player's side gets one fewer): 63 = 64 soldiers, the Vita target; 0..127
AK_BOTS=63
# bot aim error in degrees (bigger = worse shots); unset = the game's own (Weapon "AIWeapon_Accuracy_Normal": 5)
#AK_BOT_AIM=5.0
# 1 = bots' weapons carry the game's bot modifiers (Weapon "Modifier_Bot_Damage": 0.3 x the damage, "Modifier_Bot_RateOfFire":
# 3 x the time between shots; vehicle guns their own hardpoint's); 0 = bots shoot like the player
AK_BOT_GAME_MODIFIERS=1
# the chance a bot takes a free vehicle of its side within 40 m when it picks where to go (0 = bots stay on foot)
AK_BOT_VEHICLES=0.6
# 1 = bots do not shoot (tests)
AK_BOT_PASSIVE=0
# vehicles at the levels' vehicle spawns (1 = on); seconds before a wreck's spawn makes a new one (unset = each spawn's own, 10 s)
AK_VEHICLES=1
#AK_VEHICLE_RESPAWN=10
# gravity (m/s^2): the game's is 10
AK_GRAVITY=10
# soldier camera: modern (over the right shoulder, the DICE Battlefront framing; the default) / classic (the game's own 7 m camera).
# Vehicles always use the game's own chase camera.  (Until there is a settings menu, this line is the switch.)
AK_CAM_STYLE=modern
# field of view of the modern camera (vertical degrees)
AK_FOV=55
# twin-stick aiming (right stick aims freely): -1 = the device's default (on: the Vita has a right stick), 0 = off (the game's own
# scheme: the right stick turns), 1 = on.  Lock-on (hold L) works either way; reloading is automatic, as in the game.
AK_TWIN_STICK=-1
# free aim: turn speed (rad/s at full tilt), response curve (1 = linear, 2 = fine near the centre), invert the vertical
AK_AIM_SPEED=3.2
AK_AIM_CURVE=2.0
AK_INVERT_Y=0
# the soldier's side in play mode: 1 (Republic / Alliance) or 2 (CIS / Empire); unset = the soldier's own faction
#AK_TEAM=1
# conquest (the game's options): tickets each team starts with (25..200), seconds between spawn waves,
# seconds a team must hold every command post to win
AK_TICKETS=100
AK_RESPAWN_DELAY=10
AK_TOTAL_LOSS_TIME=60
