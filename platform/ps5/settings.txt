# SWBF Renegade Squadron (PS5) settings: copy to /data/SWBFRS/settings.txt
# One KEY=VALUE per line; '#' starts a comment line. Every value below is the PS5 default, so deleting a line changes nothing.

# ---- picture
# internal resolution x the PSP's 480x272: 1 to 8 (8 = 3840x2176, shown as 4K)
RS_RES_SCALE=8
# 1 = exactly 16:9 (one PSP row off the top and the bottom: 3840x2160 at scale 8); 0 = the whole 480x272 with thin bars
RS_CROP_169=1
# video output: 2160, 1440 or 1080
RS_OUTPUT=2160
# FXAA anti-aliasing: 1 = on, 0 = off
RS_FXAA=1
# anisotropic filtering: 1 = off, 2, 4, 8 or 16
RS_ANISO=16
# frame rate: 30 or 60
RS_FPS=60
# vertex transform and lighting on the GPU: 1 = on (needed by everything in the next section)
RS_HW_TL=1

# ---- lighting (PS5GFXExperimental)
# sun shadows from a shadow map: 1 = on; map size in pixels; range around the camera in metres; darkness in percent
RS_SHADOWS=1
RS_SHADOW_RES=4096
RS_SHADOW_RANGE=60
RS_SHADOW_STRENGTH=50
# PBR shading (metal/roughness + normal maps) for mod textures that come with _mr / _n maps
RS_PBR=1
# ray-traced shadows and reflections: settings only so far, the passes are not in this build (0 = off)
RS_RT_SHADOWS=0
RS_RT_REFLECTIONS=0
# per-map light setup: written to /data/SWBFRS/mods/lights/<era>_<map>.txt from the game's own lights the first time a map is
# played (after this many frames), then read from there - edit a file to change that map's sun; delete it to capture again
RS_LIGHT_CAPTURE=1
RS_LIGHT_CAPTURE_FRAMES=600

# ---- modding
# texture replacements from /data/SWBFRS/mods/<mod>/textures/ (any subfolder): 1 = on
RS_MODS=1
# 1 = write every texture of the maps, characters, weapons, vehicles and menus the game loads to /data/SWBFRS/mods/dumps/
#     (maps/<map>, players/<character>, weapons/..., vehicles/..., objects/..., effects, ui/<file>), named <key>@<name>.png:
#     a dumped file put into a mod's textures folder replaces that texture; add <key>@<name>_mr.png / _n.png for PBR
RS_DUMP_ASSETS=1

# ---- performance
# 1 = renderer, game main thread and audio on a core each, the game's other threads on the rest; 0 = left to the kernel
RS_CPU_PIN=1

# ---- controls
# 1 = left stick move + strafe, right stick turn + look; 0 = the PSP's one-stick controls
RS_TWIN_STICK=1

# ---- game options (offline games only; network games keep the game's 16 players and 25..200 tickets)
# 0 = the game's original option limits
RS_GAME_PATCHES=1
# highest "Min Players (including AI)": bots fill both teams evenly (the game's own limit is 16)
RS_MAX_PLAYERS=128
# lowest and highest "Conquest Initial Tickets" (the game's own range is 25 to 200)
RS_MIN_TICKETS=200
RS_MAX_TICKETS=5000

# ---- experimental features: 1 = on, 0 = the game's own behaviour
RS_SAVE_MANAGER=1
RS_ANY_HEAD=1
RS_COLOUR_PICKER=1
RS_ANY_SLOT=1
# load-out credit allowance (the game's is 100)
RS_CREDITS=200

# ---- sound and diagnostics
# 0 = no sound
RS_AUDIO=1
