# SWBF Renegade Squadron (PS Vita) settings: copy to ux0:data/SWBFRS/settings.txt
# One KEY=VALUE per line; '#' starts a comment line. Every value below is the default, so deleting a line changes nothing.

# ---- picture
# internal resolution x the PSP's 480x272: 1 = 480x272 (512x512 render surfaces, as the PSP; scaled up to the screen),
# 2 = 960x544 (1024x1024 surfaces, 4x the GPU memory and fill), 3 = 1440x816 (supersampled)
RS_RES_SCALE=2
# the same for the title screen and the menus (outside a match): 1 or 2.  Matches, videos and the in-game menus use RS_RES_SCALE
RS_MENU_RES_SCALE=2
# FXAA anti-aliasing while the picture is scaled to the screen (RS_RES_SCALE=1; at 2 the picture is already the screen's size): 1 = on, 0 = off
RS_FXAA=1
# fog intensity, in percent of each map's own fog: 100 = the game's (set close for the PSP), above 100 closer, below farther
# (fog distance x (2 - value/100): 25 = 75% farther away)
RS_FOG=25
# frame rate: 30 or 60
RS_FPS=60
# video memory for textures (MB), and the texture cache size before old textures are evicted (MB)
RS_TEX_POOL_MB=48
RS_TEX_BUDGET_MB=40
# vertex transform and lighting on the GPU: 1 = on
RS_HW_TL=1

# ---- performance
# 1 = renderer on core 0, game main thread on core 1, other game threads on core 2
RS_CPU_PIN=1
# 1 = use the 4th CPU core when the CapUnlocker plugin is installed (sound threads move there); 0 = 3 cores only
RS_CORE3=1
# CPU clock the game raises the Vita to at start (MHz); a higher clock already set (an overclocking plugin) is kept; 0 = leave the clocks alone
RS_ARM_MHZ=444
# 1 = the game does not wait for the renderer at GU sync points (its next frame is prepared while this one is drawn; 0 = wait, as the PSP)
RS_GU_PIPELINE=1
# lifted game functions (native rewrites of the hottest game code: collision ray casts so far); 0 = the original code, to compare
RS_LIFT=1

# ---- controls
# 1 = left stick move + strafe, right stick turn + look; 0 = the PSP's one-stick controls
RS_TWIN_STICK=1
# twin-stick turning acceleration, percent: how fast turning picks up when the right stick is pushed (100 = the game's; 130 = 30% faster)
RS_TURN_ACCEL=130
# look centring: after you stop looking up / down, the view is levelled again.  RS_LOOK_CENTER_DELAY: how many seconds after
# (decimals allowed; 1 = the game's own).  RS_LOOK_CENTER=0 never levels it (odd camera angles, mostly as a hero).
RS_LOOK_CENTER=1
RS_LOOK_CENTER_DELAY=1

# ---- game options (Instant Action "Game Options" limits)
# 0 = the game's original option limits
RS_GAME_PATCHES=1
# highest "Min Players (including AI)": bots fill both teams evenly (the game's own limit is 16)
RS_MAX_PLAYERS=64
# lowest and highest "Conquest Initial Tickets" (the game's own range is 25 to 200)
RS_MIN_TICKETS=25
RS_MAX_TICKETS=1000

# ---- AI
# most bots sent to hold one of their own command posts: 0 = no defenders (every squad attacks or takes posts); -1 = the game's own choice
RS_AI_DEFENDERS=0
# 1 = no bot stands about: squads the game forms but sends nowhere, squads waiting at a post their side owns, and bots the game
#     takes out of their squads and leaves with nothing to do are sent to the best post to take (checked every second)
RS_AI_IDLE_FIX=1
# seconds a bot with no squad may stand still before it is put back into a squad
RS_AI_IDLE_SECONDS=6

# ---- custom maps from ux0:data/SWBFRS/mods/maps (made with tools/asura/maptool.py install): 1 = listed in Instant Action
RS_CUSTOM_MAPS=1

# ---- experimental features: 1 = on, 0 = the game's own behaviour
# save manager: 5 profile slots to load, save and delete
RS_SAVE_MANAGER=1
# Personalisation: any head of the faction on any of its bodies
RS_ANY_HEAD=1
# Personalisation: HSV colour picker instead of the colour swatches
RS_COLOUR_PICKER=1
# Customize Load-out: any weapon in either weapon slot, explosives and special items in either slot
RS_ANY_SLOT=1
# load-out credit allowance (the game's is 100)
RS_CREDITS=200

# ---- sound and diagnostics
# 0 = no sound
RS_AUDIO=1
# 1 = frame-rate / load overlay:
#   FPS (shown) / VBLANK (game frames); GAME CPU / GE+GPU THREAD busy; HEAP;
#   per frame: SCENES (render-target switches), DRAWS, texture UPLOADS;
#   GPU WAIT (% of time the renderer waits for the GPU), GPU SYNCS per second (full stalls), RT (scenes the render target allows),
#   SLOW-RAM (GPU buffers that did not fit in video RAM)
RS_HUD=1
# size limit of ux0:data/SWBFRS/log.txt (MB)
RS_LOG_MB=4
# 1 = profiler lines every 600 frames to ux0:data/SWBFRS/perf.log, and the overlay's numbers in log.txt every 5 s ([HUD])
RS_PERF_LOG=0

# ---- AI and weapons
# most bots sent to hold one of their own command posts: 0 = no defenders (every squad attacks or takes posts); -1 = the game's own choice
RS_AI_DEFENDERS=0
# 1 = squads the game forms but sends nowhere (most of a side at the start of a match) are sent to the best post to take
RS_AI_IDLE_FIX=1
# vehicles and soldiers: 1 = friendly soldiers do not hold a vehicle up, an enemy soldier hit by a vehicle with someone aboard dies
RS_VEHICLE_RUNOVER=1
# Weapons_Mod.txt next to this file: changes to the game's weapons and new weapons (with on-screen names), in sections per
# faction; read at every start.  Weapons_<CIS|Republic|Imperial|Rebels|HV|All|Vehicles>_Default.txt: the game's own values,
# rewritten at every start for reference.  0 = the game's own weapons (Weapons_Mod.txt not read)
RS_WEAPONS=1
# any head on any body: the CIS too (off: the droids' heads do not fit each other's bodies)
RS_ANY_HEAD_CIS=0
# movies (the intro logos, the cut-scenes): played from MP4.  The first start converts the game's PMF movies to MP4 (H.264 + AAC)
# and its AT3 sound to WAV, into converted/ next to this file; a replacement movie goes in mods/movies/<NAME>.mp4 (NAME as the
# game's file, e.g. OPENING_LOGOS.mp4, INTRO.mp4), in the Vita movie format: 960x544 or smaller, H.264 Main profile level 3.1,
# 30 fps, 5000 kbps (8000 peak, 2-pass VBR), YUV 4:2:0 8-bit, AAC-LC 128-192 kbps, MP4 (tools/movies/vita_movie.bat makes one).
# A replacement outside that format is not played (log.txt says why): the game's own movie is.
# 0 = no movies
RS_MOVIES=1

# ---- online (Multiplayer > Infrastructure: the RS server, tools/rsserver)
# The game looks for an RS server on the local network first; if none answers it uses this address (a name or an IP).
RS_SERVER=
# 1 = look on the local network first, 0 = only RS_SERVER
RS_LAN_SEARCH=1
# diagnostics: log every frame that takes longer than this many milliseconds, with the files opened during it (0 = off)
RS_SPIKE_MS=0
