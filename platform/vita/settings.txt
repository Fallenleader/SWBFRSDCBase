# SWBF Renegade Squadron (PS Vita) settings: copy to ux0:data/SWBFRS/settings.txt
# One KEY=VALUE per line; '#' starts a comment line. Every value below is the default, so deleting a line changes nothing.

# ---- picture
# internal resolution x the PSP's 480x272: 1 = 480x272 (512x512 render surfaces, as the PSP; scaled up to the screen),
# 2 = 960x544 (1024x1024 surfaces, 4x the GPU memory and fill), 3 = 1440x816 (supersampled)
RS_RES_SCALE=1
# the same for the title screen and the menus (outside a match): 1 or 2.  Matches, videos and the in-game menus use RS_RES_SCALE
RS_MENU_RES_SCALE=2
# FXAA anti-aliasing while the picture is scaled to the screen: 1 = on, 0 = off
RS_FXAA=1
# fog intensity, in percent of each map's own fog: 100 = the game's (set close for the PSP), above 100 closer, below farther
# (fog distance x (2 - value/100): 25 = 75% farther away)
RS_FOG=25
# frame rate: 30 or 60
RS_FPS=30
# video memory for textures (MB), and the texture cache size before old textures are evicted (MB)
RS_TEX_POOL_MB=48
RS_TEX_BUDGET_MB=40
# vertex transform and lighting on the GPU: 1 = on
RS_HW_TL=1

# ---- performance
# 1 = renderer on core 0, game main thread on core 1, other game threads on core 2
RS_CPU_PIN=1
# 1 = experimental: the game does not wait for the renderer at GU sync points
RS_GU_PIPELINE=0

# ---- controls
# 1 = left stick move + strafe, right stick turn + look; 0 = the PSP's one-stick controls
RS_TWIN_STICK=1
# twin-stick turning acceleration, percent: how fast turning picks up when the right stick is pushed (100 = the game's; 130 = 30% faster)
RS_TURN_ACCEL=130

# ---- game options (Instant Action "Game Options" limits)
# 0 = the game's original option limits
RS_GAME_PATCHES=1
# highest "Min Players (including AI)": bots fill both teams evenly (the game's own limit is 16)
RS_MAX_PLAYERS=64
# highest "Conquest Initial Tickets" (the game's own limit is 200)
RS_MAX_TICKETS=1000

# ---- experimental features: 1 = on, 0 = the game's own behaviour
# save manager: 5 profile slots to load, save and delete
RS_SAVE_MANAGER=1
# Personalisation: any head of the faction on any of its bodies
RS_ANY_HEAD=1
# Personalisation: HSV colour picker instead of the colour swatches
RS_COLOUR_PICKER=1
# Customize Load-out: any weapon in either weapon slot, explosives and special items in either slot
RS_ANY_SLOT=1
# load-out credit allowance (the game's is 100)
RS_CREDITS=200

# ---- sound and diagnostics
# 0 = no sound
RS_AUDIO=1
# 1 = frame-rate / load overlay:
#   FPS (shown) / VBLANK (game frames); GAME CPU / GE+GPU THREAD busy; HEAP;
#   per frame: SCENES (render-target switches), DRAWS, texture UPLOADS;
#   GPU WAIT (% of time the renderer waits for the GPU), GPU SYNCS per second (full stalls), RT (scenes the render target allows),
#   SLOW-RAM (GPU buffers that did not fit in video RAM)
RS_HUD=1
# size limit of ux0:data/SWBFRS/log.txt (MB)
RS_LOG_MB=4
# 1 = profiler lines every 600 frames to ux0:data/SWBFRS/perf.log
RS_PERF_LOG=0

# ---- AI and weapons
# most bots sent to hold one of their own command posts: 0 = no defenders (every squad attacks or takes posts); -1 = the game's own choice
RS_AI_DEFENDERS=0
# 1 = squads the game forms but sends nowhere (most of a side at the start of a match) are sent to the best post to take
RS_AI_IDLE_FIX=1
# vehicles and soldiers: 1 = friendly soldiers do not hold a vehicle up, an enemy soldier hit by a vehicle with someone aboard dies
RS_VEHICLE_RUNOVER=1
# Weapons_Mod.txt next to this file: changes to the game's weapons and new weapons (with on-screen names), in sections per
# faction; read at every start.  Weapons_<CIS|Republic|Imperial|Rebels|HV|All|Vehicles>_Default.txt: the game's own values,
# rewritten at every start for reference.  0 = the game's own weapons (Weapons_Mod.txt not read)
RS_WEAPONS=1
# any head on any body: the CIS too (off: the droids' heads do not fit each other's bodies)
RS_ANY_HEAD_CIS=0
